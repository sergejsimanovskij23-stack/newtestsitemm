# Установка Node.js для Windows
# Скачай и установи Node.js 18 или 20 LTS:
# https://nodejs.org/ru/download/
#
# После установки перезапусти терминал и выполни:
# node --version  (должно быть v18.x или v20.x)

# Установка проекта:
npm install
npm run dev

# Для сборки:
npm run build
